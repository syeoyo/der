\subsection{Nomenclature}
\begin{itemize}[leftmargin=*]
    \item Sets and Indices:
    \begin{itemize}[leftmargin=*]\normalsize
        \item $I$: Set of individual DERs $i \in I = \{1,2,\ldots,N^I\}$
        \item $T$: Set of elementary time periods (hours) $t \in T = \{1,2,\ldots,N^T \}$
        \item $S$: Set of scenarios used to represent the evolution of uncertain parameters $s \in S = \{1,2,\ldots,N^S\}$
        \item $B$: Set of arbitrage control conditions $b \in B = \{1,2,\ldots,N^B\}$
    \end{itemize}
    \item Parameters:
    \begin{itemize}[leftmargin=*]\normalsize
        \item $P_{t}^{DA}$: Day-ahead market price at time $t$ 
        \item $P_{t}^{RT}(\xi_s)$: Real-time market price at time $t$ under scenario $s$
        \item $P_{t}^{PN}(\xi_s)$: Commitment shortage penalty price at time $t$ under scenario $s$
        \item $P_{t}^{IN}(\xi_s)$: Internal pool market price at time $t$ under scenario $s$
        \item $R_{it}(\xi_s)$: Amount of renewable energy generation by DER $i$ at time $t$ under scenario $s$
        \item $K_{i}$: Maximum level of DER $i$'s energy storage
        \item $\kappa_{i}^C, \kappa_{i}^D$: Charging and discharging rate of DER $i$'s energy storage
        \item $\eta_{i}^C, \eta_{i}^D$: Charging and discharging inefficiency of DER $i$'s energy storage
        \item $M_{it}(\xi_s)$: Maximum energy trading availability of DER $i$ at time $t$ under scenario $s$
        \item $M'_{it}(\xi_s)$: Maximum energy storage utilization of DER $i$ at time $t$ under scenario $s$
        \item $\xi_s$: Realization of uncertain parameters under scenario $s$ 
        \item $\pi_s$: Probability of occurrence of scenario $s$ 
    \end{itemize}
    \item Decision Variables:
    \begin{itemize}[leftmargin=*]\normalsize
        \item $x_{it}$: Day-ahead commitment quantity of DER $i$ at time $t$
        \item $\alpha_{it}, \beta_{it}, \gamma_{it}$: Linear decision rule coefficients for charging decisions of DER $i$ at time $t$
        \item $\alpha_{it}', \beta_{it}', \gamma_{it}'$: Linear decision rule coefficients for discharging decisions of DER $i$ at time $t$
        \item $y^{+}_{it}(\xi_s)$: Real-time market selling quantity of DER $i$ at time $t$ under scenario $s$
        \item $y^{-}_{it}(\xi_s)$: Commitment shortage quantity subject to penalty of DER $i$ at time $t$ under scenario $s$
        \item $z^{C}_{it}(\xi_s)$: Charging quantity of DER $i$ at time $t$ under scenario $s$
        \item $z^{D}_{it}(\xi_s)$: Discharging quantity of DER $i$ at time $t$ under scenario $s$
        \item $z_{it}(\xi_s)$: Energy storage level of DER $i$ at time $t$ under scenario $s$
        \item $d^{+}_{it}(\xi_s)$: Internal pool selling quantity of DER $i$ at time $t$ under scenario $s$
        \item $d^{-}_{it}(\xi_s)$: Internal pool buying quantity of DER $i$ at time $t$ under scenario $s$
        \item $\lambda_{t}(\xi_s)$: Lagrangian multiplier for internal pool supply-demand clearance at time $t$ under scenario $s$
        \item $\phi_{it}^{b}(\xi_s)$: Binary variable for arbitrage control condition $b$ for DER $i$ at time $t$ under scenario $s$
        \item $\mu_{it}^{b}(\xi_s), \delta_{it}^{b}(\xi_s)$: Auxiliary variables for convex hull relaxation of condition $b$ for DER $i$ at time $t$ under scenario $s$
